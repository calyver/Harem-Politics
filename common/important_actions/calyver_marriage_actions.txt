action_too_few_spouses = {
	priority = 900
	check_create_action = {
		if = {
			limit = {
				hp_accepts_polygamy = yes # Check if it also works for the tradition. If they don't lose piety, then it should only apply to the faith parameter
				allowed_more_spouses = yes
				is_adult = yes
				trigger_if = {
					limit = {
						primary_title.tier >= 5
					}
					any_spouse = {
						count < 4
					}
				}
				trigger_else_if = {
					limit = {
						primary_title.tier >= 4
					}
					any_spouse = {
						count < 3
					}
				}
				trigger_else_if = {
					limit = {
						primary_title.tier >= 3
					}
					any_spouse = {
						count < 2
					}
				}
				trigger_else = {
					any_spouse = {
						count < 1
					}
				}
				NOT = {
					has_government = landless_adventurer_government
				}
			}
			try_create_important_action = {
				important_action_type = action_too_few_spouses
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = marry_off_interaction
			actor = root
			recipient = root
		}
	}
}

can_visit_harem_noification = {
	priority = 899
	check_create_action = {
		if = {
			limit = {
				has_character_flag = visit_harem_decision_notification
				can_execute_decision = check_harem_decision
				any_consort = {
					is_courtier_of = root
					is_available_healthy_adult = yes
					trigger_if = {
						limit = {
							is_travelling = yes
						}
						current_travel_plan ={
							travel_plan_owner = root
						}
					}
					count >= 3
				}
				NOR = {
					has_trait_with_flag = illness				

					has_variable = flag_visited_harem
					has_variable = flag_is_visiting_harem
					
					has_character_modifier = isolating_modifier
				}
			}
			try_create_important_action = {
				important_action_type = can_visit_harem_noification
			}
		}
	}

	effect = {
		decision:check_harem_decision = { open_view_data = decision_detail }
	}
}
