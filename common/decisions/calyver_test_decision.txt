			localization = test_culture_tradition_monogamous_decision

			is_shown = {
				NOT = {
					has_cultural_tradition = tradition_concubines
				}
			}
			current_description = test_culture_tradition_concubines_decision_desc
			localization = test_culture_tradition_concubines_decision
			icon = "gfx/interface/icons/marriage.dds"
			ai_chance = { value = 0 }
		}
		item = { # Polygamous
			value = tradition_polygamous
			is_shown = {
				NOT = {
					has_cultural_tradition = tradition_polygamous
				}
			}

			localization = test_culture_tradition_polygamous_decision

			ai_chance = { value = 0 }

		item = { # Harem

			ai_chance = { value = 0 }

		item = { # Cosmic Order
			value = tradition_cosmic_order
			is_shown = {
				NOT = {
					has_cultural_tradition = tradition_cosmic_order
				}
			}
			current_description = test_culture_tradition_cosmic_order_decision_desc
			localization = test_culture_tradition_cosmic_order_decision
			icon = "gfx/interface/icons/marriage.dds"
			ai_chance = { value = 0 }
		}
		item = { # Court Machinations
			value = tradition_tgp_court_machinations
			is_shown = {
				NOT = {
					has_cultural_tradition = tradition_tgp_court_machinations
				}
			}
			current_description = test_culture_tradition_tgp_court_machinations_decision_desc
			localization = test_culture_tradition_tgp_court_machinations_decision
			icon = "gfx/interface/icons/marriage.dds"
			ai_chance = { value = 0 }
		}
		item = { # Japanese Houses
			value = tradition_tgp_japanese_houses
			is_shown = {
				NOT = {
					has_cultural_tradition = tradition_tgp_japanese_houses
				}
			}
			current_description = test_culture_tradition_tgp_japanese_houses_decision_desc
			localization = test_culture_tradition_tgp_japanese_houses_decision
			icon = "gfx/interface/icons/marriage.dds"
			ai_chance = { value = 0 }
		}
	}

	effect = {
		culture = {
			calyver_remove_culture_traditions_test_effect = yes
			hidden_effec = {
				if = {
					limit = {
						scope:tradition_monogamous ?= yes
					}
					add_culture_tradition = tradition_monogamous
				}
				else_if = {
					limit = {
						scope:tradition_concubines ?= yes
					}
					add_culture_tradition = tradition_concubines
				}
				else_if = {
					limit = {
						scope:tradition_polygamous ?= yes
					}
					add_culture_tradition = tradition_polygamous
				}
				else_if = {
					limit = {
						scope:tradition_harem ?= yes
					}
					add_culture_tradition = tradition_harem
				}
				else_if = {
					limit = {
						scope:tradition_cosmic_order ?= yes
					}
					add_culture_tradition = tradition_cosmic_order
				}
				else_if = {
					limit = {
						scope:tradition_tgp_court_machinations ?= yes
					}
					add_culture_tradition = tradition_tgp_court_machinations
				}
				else_if = {
					limit = {
						scope:tradition_tgp_japanese_houses ?= yes
					}
					add_culture_tradition = tradition_tgp_japanese_houses
				}
				else = {
					error_log = "No valid culture tradition selected!"
				}
			}
		}
		custom_tooltip = test_culture_tradition_decision_tooltip
	}

	ai_check_interval = 0
}

	decision_group_type = debug
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}

	is_shown = {
		debug_only = yes
		faith = {
			OR = {
				has_doctrine = doctrine_concubines_3
				has_doctrine = doctrine_concubines_10
			}
		}
	}

	effect = {
		faith = {
			if = {
				limit = {
					has_doctrine = doctrine_concubines_3
				}
				remove_doctrine = doctrine_concubines_3
			}
			else_if = {
				limit = {
					has_doctrine = doctrine_concubines_10
				}
				remove_doctrine = doctrine_concubines_10
			}
			add_doctrine = doctrine_concubines_0
		}
	}

	ai_check_interval = 0
}
