# The Silkworm Goddess Ceremony
silkworm_goddess_ceremony_decision = {
	title = silkworm_goddess_ceremony_decision
	desc = silkworm_goddess_ceremony_decision_desc
	selection_tooltip = silkworm_goddess_ceremony_decision_selection_tooltip
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}
	sort_order = 162
	cooldown = { years = 3 }
	confirm_text = silkworm_goddess_ceremony_decision_confirm

	is_shown = {
		is_male = yes
		can_have_four_consorts_trigger = yes
	}

	is_valid = {
		OR = {
			custom_description = {
				text = primary_spouse_silkworm_worship_tooltip
				primary_spouse ?= {
					is_available_healthy_adult = yes
					is_female = yes
				}
			}
			custom_description = {
				text = mother_silkworm_worship_tooltip
				mother ?= {
					is_available_healthy_adult = yes
					is_female = yes
				}
			}
		}
		custom_description = {
			text = silkworm_goddess_ceremony_month_tooltip
			current_month >= 1
			current_month <= 2
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = mourning_bride_show
			NOT = { has_character_modifier = mourner_modifier }
		}

		custom_tooltip = {
			text = contagious_disease_bride_show
			has_epidemic_disease_trigger = no
		}
		custom_tooltip = {
			text = epidemic_bride_show
			NOR = {
				realm_has_any_epidemic = {
					SIZE = 1
					INTENSITY = minor
				}
				capital_county ?= { has_county_modifier = isolate_capital_decision_modifier }
			}
		}
		custom_tooltip = {
			text = seclusion_bride_show
			NOR = {
				has_character_modifier = isolating_modifier
			}
		}
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = { has_treasury = no }
				add = medium_gold_value
			}
		}
		treasury = {
			value = 0
			if = {
				limit = { has_treasury = yes }
				add = medium_treasury_value
			}
		}
	}

	effect = {
		save_scope_as = ruler_scope
		hidden_effect = {
			if = {
				limit = {
					any_courtier = {
						is_head_of_the_harem_trigger = { LIEGE = root }
						is_available_healthy_adult = yes
						is_courtier_of = root
						is_female = yes
						OR = {
							is_parent_of = scope:ruler_scope
							scope:ruler_scope.primary_spouse ?= this
						}
					}
				}
				random_courtier = {
					limit = {
						is_head_of_the_harem_trigger = { LIEGE = root }
					}
					save_scope_as = ritual_leader
				}
			}
			else_if = {
				limit = {
					mother ?= {
						is_available_healthy_adult = yes
						is_courtier_of = root
					}
				}
				mother = { save_scope_as = ritual_leader }
			}
			else_if = {
				limit = {
					primary_spouse ?= {
						is_female = yes
						is_available_healthy_adult = yes
						is_courtier_of = root
					}
				}
				primary_spouse = {
					save_scope_as = ritual_leader
				}
			}
		}
		trigger_event = {
			id = harem_events.2000
			days = { 20 35 }
		}
	}
	
	ai_potential = { always = yes }

	ai_check_interval = 36
	ai_will_do = {
		base = 30
		ai_value_modifier = {
			ai_zeal = 0.25
		}
		modifier = {
			is_valid_for_legitimacy_change = yes
			legitimacy_level < 6
			add = 5
		}
		modifier = {
			is_valid_for_legitimacy_change = yes
			legitimacy_level < 5
			add = 10
		}
		modifier = {
			is_valid_for_legitimacy_change = yes
			legitimacy_level < 4
			add = 10
		}
		modifier = {
			any_courtier = {
				is_head_of_the_harem_trigger = { LIEGE = root }
				is_available_healthy_adult = yes
				is_courtier_of = root
				is_female = yes
				OR = {
					is_parent_of = root
					root.primary_spouse ?= this
				}
				learning >= decent_skill_rating
			}
			add = 30
		}
		modifier = {
			trigger_if = {
				limit = {
					has_treasury = no
				}
				gold < root.major_gold_value
			}
			trigger_else = {
				treasury < root.major_treasury_value
			}
			factor = 0
		}
	}
}

	}
	sort_order = 162
	cooldown = { years = 3 }
	confirm_text = visit_matriarch_decision_confirm

	is_shown = {
		OR = {
			hp_has_seniority_based_harem_relatives_trigger = yes
			religion = religion:islam_religion
		}
		any_courtier = {
			is_eligible_harem_member_trigger = {
				LIEGE = root
			}
			trigger_if = {
				limit = {

							is_eligible_harem_member_trigger = {
								LIEGE = root
							}

					}

			}

			}

		custom_tooltip = {
			text = contagious_disease_bride_show
			has_epidemic_disease_trigger = no
		}
