						days = 2
					}
					if = { #New Primary Spouse
						limit = {
							any_spouse = {
								NOT = {
									this = scope:dead_relation
								}
								count >= 1
							}
						}
						new_primary_spouse_on_action_effect = yes
					}
				}
			}

				has_game_rule = mourning_cycle

				}

					}
					add_to_list = mourners_list
				}
			}

				limit = {
					any_child = {
						is_ruler = yes
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
						count >= 1
					}
				}

					limit = {
						is_ruler = yes
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
					}
					add_character_flag = {
						flag = dead_important_relative_flag
						days = 2
					}
					add_to_list = mourners_list
				}
			}

				limit = {
					age >= 10
					any_parent = {
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
						highest_held_title_tier >= tier_duchy
						is_ruler = yes
						player_heir ?= scope:dead_relation
						count >= 1
					}
				}

					limit = {
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
						highest_held_title_tier >= tier_duchy

					}

						flag = dead_important_relative_flag
						days = 2

				}

		# 	}

		# if = {

		if = {

						if = {
							limit = {

						}

							}

						}
						if = {
							limit = {
								is_clergy = yes
							}
							add = 10
						}
					}
					max = 1
					check_range_bounds = no
					save_scope_as = temple_liege
				}
			}
			every_in_list = {
				list = dowager_consort_list
				trigger_event = {
					on_action = calyver_imperial_consort_management_on_action
					days = 1
				}
			}
		}
	}

calyver_imperial_consort_management_on_action = {
	effect = {
		if = {
			limit = {
				has_trait = celibate
			}
			remove_trait = celibate
		}
		# Once an imperial consort, always an imperial consort
		if = {
			limit = {
				has_character_flag = devoted_widow_flag
			}
			add_trait = devoted_widow
			remove_character_flag = devoted_widow_flag
			if = { # Parents go to their children
				limit = {
					court_owner ?= {
						NOT = {
							is_child_of = prev
						}
					}
					any_child = {
						is_child_of = scope:dead_relation
						highest_held_title_tier >= tier_barony
						is_landed_or_landless_administrative_or_theocratic = yes
					}
				}
				ordered_child = {
					limit = {
						is_child_of = scope:dead_relation
						highest_held_title_tier >= tier_barony
						is_landed_or_landless_administrative_or_theocratic = yes
					}
					order_by = highest_held_title_tier
					max = 1
					check_range_bounds = no
					add_courtier = prev
				}
			}
		}
		else = {
			add_trait = devoted
			# Send them to the imperial nunnery
			if = {
				limit = {
					exists = scope:temple_liege
				}
				# Check if this court already identified the nunnery holder
				scope:temple_liege = {
					add_courtier = prev
				}
			}
		}
	}
}
