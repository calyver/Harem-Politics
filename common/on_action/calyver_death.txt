calyver_on_death_on_action = {
	effect = {
		save_scope_as = dead_relation
		if = { #Spouses
			limit = {
				is_married = yes
			}
			every_spouse = {
				limit = {
					is_ruler = yes
				}
				if = { #Primary Spouse
					limit = {
						primary_spouse ?= scope:dead_relation
					}
					add_character_flag = {
						flag = dead_main_spouse_flag
						days = 2
					}
					if = { #New Primary Spouse
						limit = {
							any_spouse = {
								NOT = {
									this = scope:dead_relation
								}
								count >= 1
							}
						}
						new_primary_spouse_on_action_effect = yes
					}
				}
			}
		}
		if = {
			limit = {
				has_game_rule = mourning_cycle
			}
			if = { #Spouses
				limit = {
					is_married = yes
				}
				every_spouse = {
					limit = {
						is_ruler = yes
					}
					add_to_list = mourners_list
				}
			}
			if = { #Children
				limit = {
					any_child = {
						is_ruler = yes
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
						count >= 1
					}
				}
				every_child = {
					limit = {
						is_ruler = yes
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
					}
					add_character_flag = {
						flag = dead_important_relative_flag
						days = 2
					}
					add_to_list = mourners_list
				}
			}
			if = { #Parents
				limit = {
					age >= 10
					any_parent = {
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
						highest_held_title_tier >= tier_duchy
						is_ruler = yes
						player_heir ?= scope:dead_relation
						count >= 1
					}
				}
				every_parent = {
					limit = {
						NOR = {
							has_trait = incapable
							is_in_list = mourners_list
						}
						highest_held_title_tier >= tier_duchy
						is_ruler = yes
						player_heir ?= scope:dead_relation
					}
					add_character_flag = {
						flag = dead_important_relative_flag
						days = 2
					}
					add_to_list = mourners_list
				}
			}
			every_in_list = {
				list = mourners_list
				trigger_event = {
					id = harem_politics.0015
					days = 1
				} 
			}
		}
		# Set up the variable lists to prevent incest
		# every_consort = {

		# 	add_to_temporary_list = former_consorts_list
		# }

		# 	limit = {
		# 		is_ruler = no
		# 		NOT = { is_in_list = former_consorts_list }
		# 	}
		# 	add_to_temporary_list = former_consorts_list
		# }
		# every_former_concubine = {
		# 	limit = {
		# 		is_ruler = no
		# 		NOT = { is_in_list = former_consorts_list }
		# 	}
		# 	add_to_temporary_list = former_consorts_list
		# }
		# if = {
		# 	limit = {
		# 		any_in_list = {
		# 			list = former_consorts_list
		# 			count > 0
		# 		}
		# 	}

		# Add the Devoted Widow trait
		if = {
			limit = {
				can_have_retired_imperial_consorts = yes
				any_consort = {
					should_be_retired_imperial_consort_trigger = yes
				}
			}
			every_consort = {
				limit = {
					should_be_retired_imperial_consort_trigger = yes
				}
				if = {
					limit = {
						OR = {
							is_spouse_of = scope:dead_relation
							any_child = {
								even_if_dead = yes

							}

						}
					}

calyver_imperial_consort_management_on_action = {
	effect = {
		if = {
			limit = {
				has_trait = celibate
			}
			remove_trait = celibate
		}
		# Once an imperial consort, always an imperial consort
		if = {
			limit = {
				has_character_flag = devoted_widow_flag
			}
			add_trait = devoted_widow
			remove_character_flag = devoted_widow_flag
		}
		else = {
			add_trait = devoted
		}
	}
}
