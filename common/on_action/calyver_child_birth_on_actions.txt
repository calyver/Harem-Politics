# On actions for child birth and pregnancy notification events, triggered automatically by the game

# called for the newborn child
# scope:child is the new born character
# scope:mother is the pregnant mother
# scope:father is the family father (not the biological)
# scope:real_father is the biological father
# scope:is_bastard, true if a known bastard
on_birth_child = {
	first_valid_on_action = {
		calyver_on_birth_child_on_actions
	}
}
calyver_on_birth_child_on_actions = {
	# events = {
	# 	birth.1180 # Primary Child
	# }
	effect = {
		if = {
			limit = {
				any_parent = {
					is_dishu_parent_trigger = yes
					save_temporary_scope_as = dishu_parent
				}
				should_be_di_child_trigger = {
					PARENT = scope:dishu_parent
				}
			}
			add_trait = di_child
			if = {
				limit = {
					mother ?= {
						is_alive = yes
						has_character_flag = should_be_di_child_flag
					}
				}
				mother = {
					remove_character_flag = should_be_di_child_flag
				}
			}
		}
	}
}
