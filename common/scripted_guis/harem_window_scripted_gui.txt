toggle_harem_window = {
	scope = character
    effect = {
		calyver_toggle_harem_window_effect = yes
    }
}

visit_harem_gui = {
	scope = character 
	effect = {
		visit_harem_effect = yes
	}
}

refresh_harem_window = {
	scope = character
	effect = {
		calyver_update_harem_list_effect = yes
	}
}

toggle_visit_harem_notification = {
	scope = character
	effect = {
		if = {
			limit = {
				has_variable = visit_harem_decision_notification
			}
			remove_variable = visit_harem_decision_notification
		}
		else = {
			set_variable = visit_harem_decision_notification
		}
	}
}
