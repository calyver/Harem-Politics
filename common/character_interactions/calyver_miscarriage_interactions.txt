miscarriage_interaction = {
	icon = pregnant
	common_interaction = yes
	interface_priority = 150
	category = interaction_category_hostile

	ignores_pending_interaction_block = yes

	desc = miscarriage_interaction_desc

	is_available = {
		OR = {
			is_married = yes
			is_concubine = yes
		}
		is_physically_able = yes
		trigger_if = {
			limit = { is_ai = yes }
			NOR = {
				has_trait = compassionate
				has_trait = forgiving
				has_trait = honest
				has_trait = just
			}
		}
	}

	is_shown = {
		scope:recipient != scope:actor
		scope:recipient = {
			NOT = {
				is_consort_of = scope:actor
			}
			in_diplomatic_range = scope:actor
			is_pregnant = yes
			has_trait = pregnant
			pregnancy_month <= 8
			pregnancy_assumed_father ?= {
				is_ruler = yes
				highest_held_title_tier >= tier_county
				OR = {
					any_courtier = {
						this = scope:actor
					}
					any_vassal_or_below = {
						this = scope:actor
					}
				}
				any_consort = {
					this = scope:actor
				}
				any_consort = {
					this = scope:recipient
				}
				OR = {
					any_courtier = {
						this = scope:recipient
					}
					any_vassal_or_below = {
						this = scope:recipient
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOR = {
				has_any_best_good_relationship_with_character_trigger = { CHARACTER = scope:recipient }
				has_any_good_relationship_with_character_trigger = { CHARACTER = scope:recipient }
			}
		}
		custom_tooltip = {
			scope:actor = {
				NOT = {
					has_character_flag = attempted_miscarriage_flag
				}
			}
			text = recently_attempted_miscarriage_message
		}
	}
	
	on_accept = {
		scope:actor = {
			trigger_event = consort_events.0020
		}
	}
	
	ai_targets = {
		ai_recipients = realm_characters
	}
	
	ai_frequency = 15
	ai_will_do = {
		base = 0
		ai_value_modifier = {
			ai_boldness = 0.15
			ai_honor = -0.2
		}
		modifier = { #New consorts are less likely to do so
			add = -10
			scope:actor = { has_character_flag = flag_new_consort }
		}
		modifier = { #Evil consorts
			add = 15
			scope:actor = { harem_politics_evil_consort_trigger = yes }
		}
		modifier = { #Rivals
			add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:recipient }
			}
		}
		modifier = { #Scheming consorts
			add = 15
			scope:actor = { has_character_modifier = scheming_consort_modifier }
		}
		opinion_modifier = { #Opinion
			who = scope:actor
			opinion_target = scope:recipient
			multiplier = -0.3
		}
		modifier = { #Favored consorts
			add = 15
			scope:recipient = {
				has_character_modifier = favored_consort_modifier
			}
		}
		modifier = { #Favored consorts
			add = 25
			scope:recipient = {
				has_character_modifier = beloved_consort_modifier
			}
		}
		modifier = { #Favored consorts
			add = 35
			scope:recipient = {
				has_character_modifier = favor_boost_modifier
			}
		}
		modifier = { #The main spouse must protect their position
			add = 20
			scope:actor = {
				is_married = yes
				any_spouse = {
					primary_spouse = scope:actor
					this = scope:recipient.pregnancy_assumed_father
					save_temporary_scope_as = temp_liege
				}
			}
			scope:recipient = {
				OR = {
					any_close_family_member = {
						OR = {
							is_powerful_vassal_of = scope:temp_liege
							is_councillor_of = scope:temp_liege
							is_parent_of = scope:temp_liege
						}
					}
					is_ranked_consort_trigger = {
						LIEGE = scope:temp_liege
					}
					is_spouse_of = scope:temp_liege
				}
			}
		}
		modifier = { #The Grand Consort must protect their position
			add = 15
			scope:actor = {
				has_high_consort_position_trigger = yes
				court_owner ?= {
					is_consort_of = scope:actor
					this = scope:recipient.pregnancy_assumed_father
					save_temporary_scope_as = temp_liege
				}
			}
			scope:recipient = {
				OR = {
					any_close_family_member = {
						OR = {
							is_powerful_vassal_of = scope:temp_liege
							is_councillor_of = scope:temp_liege
							is_parent_of = scope:temp_liege
						}
					}
					is_ranked_consort_trigger = {
						LIEGE = scope:temp_liege
					}
					is_spouse_of = scope:temp_liege
				}
			}
		}
		modifier = { #The ICs must protect their position
			add = 15
			scope:actor = {
				has_imperial_concubine_position_trigger = yes
				court_owner ?= {
					is_consort_of = scope:actor
					this = scope:recipient.pregnancy_assumed_father
					save_temporary_scope_as = temp_liege
				}
			}
			scope:recipient = {
				OR = {
					any_close_family_member = {
						OR = {
							is_powerful_vassal_of = scope:temp_liege
							is_councillor_of = scope:temp_liege
							is_parent_of = scope:temp_liege
						}
					}
					is_ranked_consort_trigger = {
						LIEGE = scope:temp_liege
					}
					is_spouse_of = scope:temp_liege
				}
			}
		}
		modifier = { # Spouses must protect their position
			add = 15
			scope:actor = {
				any_spouse = {
					primary_spouse ?= {
						this != scope:actor
						this != scope:recipient
					}
					this = scope:recipient.pregnancy_assumed_father
					save_temporary_scope_as = temp_liege
				}
			}
			scope:recipient = {
				OR = {
					any_close_family_member = {
						OR = {
							is_powerful_vassal_of = scope:temp_liege
							is_councillor_of = scope:temp_liege
							is_parent_of = scope:temp_liege
						}
					}
					is_ranked_consort_trigger = {
						LIEGE = scope:temp_liege
					}
					is_spouse_of = scope:temp_liege
				}
			}
		}
		modifier = { #Intrigue
			add = {
				value = scope:actor.intrigue
			}
			scope:actor.intrigue > medium_skill_rating
		}
		modifier = { #Intrigue
			add = -15
			scope:actor.intrigue < medium_skill_rating
		}
		modifier = { #Intrigue
			add = -20
			scope:actor.intrigue <= mediocre_skill_rating
		}
		modifier = { #Craven consorts
			add = -10
			scope:actor = {
				has_trait = craven
			}
		}
		modifier = { #Shy consorts
			add = -10
			scope:actor = {
				has_trait = shy
			}
		}
		modifier = { #Paranoid consorts
			add = 10
			scope:actor = {
				has_trait = paranoid
			}
		}
		modifier = { #Brave consorts
			add = 15
			scope:actor = {
				has_trait = brave
			}
		}
		modifier = { #Advanced pregnancies
			add = -20
			scope:actor = {
				ai_rationality > 0
			}
			scope:recipient = {
				pregnancy_month >= 7
			}
		}
		modifier = { #Advanced pregnancies
			add = -20
			scope:actor = {
				ai_rationality > 0
			}
			scope:recipient = {
				pregnancy_month >= 4
			}
		}
	}

	auto_accept = yes
}
