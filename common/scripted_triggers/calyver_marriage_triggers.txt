# Is the TAKER (scope:puppet_or_actor) blocked from taking concubines by mourning?
# ROOT = irrelevant; this checks the taker in $CHARACTER$ param
# Returns yes = blocked, no = allowed
mourning_blocks_taker_from_concubinage_trigger = {
	$CHARACTER$ = {
		has_character_modifier = mourner_modifier
	}
}

# Is ROOT (the potential concubine) blocked from being taken as a concubine by $TAKER$ due to mourning?
# ROOT = potential concubine
# $TAKER$ = the character who would take/receive the concubine
# Returns yes = blocked, no = allowed
#
# RULES:
# - A mourning character who is imprisoned by a FOREIGN ruler can still be taken (captor doesn't care)
# - A mourning character who is your OWN courtier cannot be taken during mourning
# - A mourning character who was formerly a consort of the taker's parent cannot be taken during mourning
# - A non-mourning character is never blocked by this trigger
mourning_blocks_target_from_concubinage_trigger = {
	trigger_if = {
		limit = {
			has_character_modifier = mourner_modifier
		}
		OR = {
			# Block: they are your own courtier (you respect your court's mourning)
			is_courtier_of = $TAKER$
			# Block: they are a former consort of the taker's parent (parent's widow protection)
			OR = {
				any_former_spouse = {
					even_if_dead = yes
					any_child = {
						this = $TAKER$
					}
				}
				any_former_concubinist = {
					even_if_dead = yes
					any_child = {
						this = $TAKER$
					}
				}
			}
		}
	}
	# If they're mourning BUT imprisoned by a foreign ruler → NOT blocked (trigger passes)
	# If they're not mourning → NOT blocked (trigger_if doesn't fire, passes)
}

make_concubine_is_valid_trigger = {
	# Actor must be allowed more concubines
	custom_description = {
		text = take_concubine_allowed_more
		object = scope:recipient
		scope:puppet_or_actor = {
			allowed_more_concubines = yes
		}
	}
	# Actor must not be mourning
	scope:puppet_or_actor = {
		custom_tooltip = {
			text = take_concubine_mourning_actor
			NOT = { has_character_modifier = mourner_modifier }
		}
	}
	scope:recipient = {
		# Marriage status: either imprisoned + married (but not your spouse), or not married
		OR = {
			AND = {
				is_imprisoned_by = scope:puppet_or_actor
				is_married = yes
			}
			is_married = no
		}
		is_hostage = no
		# Mourning check on the target (concubine)
		custom_tooltip = {
			text = take_concubine_mourning_target
			NOT = {
				mourning_blocks_target_from_concubinage_trigger = { TAKER = scope:puppet_or_actor }
			}
		}
		can_be_taken_as_muslim_concubine_trigger = { 
			CHARACTER = scope:puppet_or_actor 
			RECEIVER = scope:puppet_or_actor
		}
	}
}

can_become_concubine_of_character_trigger = {
	is_available_quick = {
		adult = yes
		ruler = no
		advanced_ruler = no
		hostage = no
	}
	trigger_if = {
		limit = {
			NOT = { is_imprisoned_by = $CHARACTER$ }
		}
		is_married = no
		is_betrothed = no
		is_concubine = no
		number_of_concubines = 0
		# Mourning: a free mourning courtier cannot be offered as a concubine
		NOT = { has_character_modifier = mourner_modifier }
	}
	trigger_if = {
		limit = {
			is_imprisoned_by = $CHARACTER$
			is_married = yes
		}
		NOT = { is_spouse_of = $CHARACTER$ }
	}
	# Mourning: block if the target is a parent's widow of the taker
	trigger_if = {
		limit = {
			has_character_modifier = mourner_modifier
		}
		NOR = {
			any_former_spouse = {
				even_if_dead = yes
				any_child = {
					this = $CHARACTER$
				}
			}
			any_former_concubinist = {
				even_if_dead = yes
				any_child = {
					this = $CHARACTER$
				}
			}
		}
	}
	NOR = {
		is_concubine_of = $CHARACTER$
		has_character_flag = has_been_offered_as_concubine
	}
	could_marry_character_trigger = { CHARACTER = $CHARACTER$ } #Gender, recent divorce, allowed to marry, no illegal incest etc.
	can_be_taken_as_muslim_concubine_trigger = { # Can a Muslim be taken as a concubine?
		CHARACTER = $CHARACTER$
		RECEIVER = $CHARACTER$
	}
}

## Harem Politics triggers ##
valid_marriage_legitimacy_rank_trigger = {
	# ROOT = character being checked (either OTHER or their relative)
	# RULER = character getting legitimacy

	OR = {
		AND = {
			is_landed = yes
			is_governor_or_admin_count = no
		}
		tgp_is_ceremonial_liege_trigger = yes
	}
	OR = {
		# Outranks RULER or the same rank
		tier_difference = {
			target = $RULER$
			value >= 0
		}
		# Special case: King/Emperor for Emperor+ (including future hegemony)
		highest_held_title_tier >= tier_kingdom
	}
}

can_be_taken_as_muslim_concubine_trigger = {
	# CHARACTER = the actor, either taking the concubine or offering them. In other words, it evaluates whether the sender could take the concubine if their stance on concubinage was relevant
	# RECEIVER = the character being offered the concubine, if they exist
	custom_tooltip = {
		text = take_muslim_concubine_trigger
		trigger_if = {
			limit = {
				religion = religion:islam_religion
			}
			OR = {
				is_imprisoned_by = $CHARACTER$ # If they are imprisoned, they kinda are slaves already
				AND = {
					is_concubine_of = $CHARACTER$ # To send concubines to fellow Muslims
					$RECEIVER$ != $CHARACTER$
				}
				$CHARACTER$ = {
					trigger_if = {
						limit = {
							religion = religion:islam_religion
						}
						has_trait = cynical
					}
					trigger_else = {
						religion != religion:islam_religion
					}
				}
				has_character_flag = abducted_non_believer
			}
			trigger_if = { # Offering a concubine to a fellow Muslim
				limit = {
					$RECEIVER$ != $CHARACTER$
				}
				trigger_if = { # If the concubine isn't a slave, the receiver can't be a Muslim, or they're cynical
					limit = {
						is_imprisoned = no
						is_concubine = no
						NOT = { has_character_flag = abducted_non_believer }
					}
					$RECEIVER$ = {
						trigger_if = {
							limit = {
								religion = religion:islam_religion
							}
							has_trait = cynical
						}
						trigger_else = {
							religion != religion:islam_religion
						}
					}
				}
			}
		}
		trigger_else = {
			religion != religion:islam_religion
		}
	}
	# You cannot offer your children's parents
	trigger_if = {
		limit = {
			$CHARACTER$ != $RECEIVER$
		}
		trigger_if = { # Prevent the AI from offering potential heirs
			limit = {
				has_trait = pregnant
				exists = pregnancy_assumed_father
			}
			NOT = { pregnancy_assumed_father = $CHARACTER$ }
		}
		trigger_if = { # After having a child, a concubine can't be given away
			limit = {
				$CHARACTER$.religion = religion:islam_religion
			}
			NOT = {
				any_child = {
					even_if_dead = yes
					is_child_of = $CHARACTER$
					count > 0
				}
			}
		}
	}
	custom_tooltip = {
		text = take_concubine_ex_spouse_faith
		trigger_if = { # You can't take a freed person as a concubine
			limit = {
				$CHARACTER$.religion = religion:islam_religion
			}
			NOR = {
				is_spouse_of = $CHARACTER$
				has_divorced_opinion_trigger = {
					TARGET = $CHARACTER$
				}
				any_former_spouse = {
					this = $CHARACTER$
				}
			}
		}
	}
	custom_tooltip = {
		text = can_not_take_ex_spouse_as_concubine_tradition
		trigger_if = {
			limit = {
				$CHARACTER$ = {
					culture = { has_cultural_parameter = dishu_system }
					NAND = { # Independent kings and emperors can take ex-spouses as concubines
						can_have_imperial_consorts_trigger = yes
						is_independent_ruler = yes
					}
				}
			}
			NOR = { # You cannot take an ex-spouse as a concubine
				is_spouse_of = $CHARACTER$
				has_divorced_opinion_trigger = {
					TARGET = $CHARACTER$
				}
				any_former_spouse = {
					this = $CHARACTER$
				}
			}
		}
	}
}

# The Li Clan stopped marrying empresses / official wives in the late Tang
is_tang_imperial_line_trigger = {
	# 1. Check that the Dynastic Cycle is currently NOT in a stable era
	situation:dynastic_cycle ?= { # Verify exact situation key in common/situations/
		NOT = { has_situation_top_phase_parameter = era_type_stable }
	}

	# 2. Target belongs to the Li (Tang) dynasty
	$CHARACTER$.dynasty ?= dynasty:li_674E_87

	# 3. Tang holds the Mandate, and target is the Emperor or top 4 dynastic heirs
	title:h_china.holder ?= {
		dynasty ?= dynasty:li_674E_87
		OR = {
			this = $CHARACTER$
			AND = {
				exists = player_heir
				player_heir_position = {
					target = $CHARACTER$
					value <= 3 # 0, 1, 2, 3 = 1st through 4th in line
				}
			}
		}
	}
}
