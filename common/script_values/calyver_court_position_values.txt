# Shared High Consort AI Position & Candidate Scoring #
high_consort_shared_ai_position_score_value = {
	value = 0
	scope:liege = {
		if = {
			limit = {
				any_consort = {
					is_valid_for_favor_system_trigger = { HAREM_LIEGE = scope:liege }
					count > 5
				}
			}
			add = 50
		}
		if = {
			limit = {
				any_consort = {
					is_valid_for_favor_system_trigger = { HAREM_LIEGE = scope:liege }
					count > 7
				}
			}
			add = 50
		}
		if = {
			limit = {
				any_consort = {
					is_valid_for_favor_system_trigger = { HAREM_LIEGE = scope:liege }
					count > 9
				}
			}
			add = 50
		}
		if = {
			limit = {
				highest_held_title_tier > tier_empire
			}
			add = 50
		}
		# There's no PS
		if = {
			limit = {
				OR = {
					is_married = no
					primary_spouse ?= {
						OR = {
							is_imprisoned = yes
							NOR = {
								is_courtier_of = scope:liege
								is_vassal_of = scope:liege
							}
						}
					}
				}
			}
			add = 50
		}
	}
	add = court_position_debt_considerations_value
}

# 1. Replaces court_position_candidate_score_base_value (exempts Imperial Concubines from the -150 penalty)
high_consort_candidate_score_base_value = {
	value = 0
	scope:liege = {
		# Not very likely to hire rivals
		if = {
			limit = {
				has_relation_rival = scope:employee
			}
			add = -1000
		}
		if = {
			limit = {
				NOT = { exists = scope:firing_court_position }
			}
			if = {
				limit = {
					scope:employee = {
						has_any_court_position = yes
						NOT = { has_court_position = imperial_concubine_court_position }
					}
				}
				add = -150
			}
			if = {
				limit = {
					OR = {
						AND = {
							government_has_flag = government_is_merchant
							monthly_character_income < merchant_ai_court_position_min_income_value
						}
						AND = {
							government_has_flag = government_is_republic
							is_landless_ruler = yes
							monthly_character_balance < merchant_ai_court_position_min_income_value
						}
					}
				}
				add = -10000
			}
		}
	}
}

# 2. Replaces court_position_candidate_aptitude_value (lets each High Consort's unique aptitude do the heavy lifting)
high_consort_candidate_aptitude_value = {
	value = 0
	scope:employee = {
		# More likely to pick a courtier with a high aptitude, less likely to pick one with low
		add = {
			value = scope:employee_aptitude
			multiply = 15 # Terrible = 0, Poor = +15, Average = +30, Good = +45, Excellent = +60
		}
		if = {
			limit = {
				NOT = { exists = scope:firing_court_position }
				scope:highest_available_aptitude > scope:employee_aptitude
			}
			add = -25
		}
		if = {
			limit = {
				scope:employee_aptitude <= 0
			}
			add = -30 # Discourages hiring Terrible candidates
		}
	}
}

# 3. Shared personal relations & favor score for ranked consorts (Four Consorts & Imperial Concubines)
consort_candidate_relations_and_favor_score_value = {
	value = 0
	scope:liege = {
		# Personal Relations
		if = {
			limit = {
				has_relation_soulmate = scope:employee
			}
			add = 60
		}
		else_if = {
			limit = {
				OR = {
					has_relation_lover = scope:employee
					has_relation_best_friend = scope:employee
				}
			}
			add = 35
		}
		else_if = {
			limit = {
				has_relation_friend = scope:employee
			}
			add = 15
		}

		# Favor (trimmed down since aptitude already rewards/penalizes favor)
		if = {
			limit = {
				scope:employee = { harem_politics_beloved_consort_trigger = yes }
			}
			add = 35
		}
		else_if = {
			limit = {
				scope:employee = { harem_politics_favored_consort_trigger = yes }
			}
			add = 20
		}
		if = {
			limit = {
				scope:employee = { has_character_modifier = favor_boost_modifier }
			}
			add = 15
		}
		if = {
			limit = {
				scope:employee = { harem_politics_unfavored_consort_trigger = yes }
			}
			add = -15
		}
		if = {
			limit = {
				scope:employee = { harem_politics_forgotten_consort_trigger = yes }
			}
			add = -25
		}
	}
}

# 4. Personal, noble birth & opinion overrides for the Four Consorts
high_consort_candidate_personal_score_value = {
	value = consort_candidate_relations_and_favor_score_value
	scope:liege = {
		# Noble birth safeguard
		if = {
			limit = {
				scope:employee = { is_lowborn = yes }
			}
			add = -15
			if = {
				limit = {
					is_spouse_of = scope:employee
				}
				add = -15
			}
		}

		# Opinion
		add = {
			value = "scope:liege.opinion(scope:employee)"
			multiply = 0.25
		}
	}
}

imperial_concubine_candidate_personal_score_value = {

		# Opinion has double the weight for Imperial Concubines compared to the Four Consorts

			multiply = 0.5
		}
	}
}
