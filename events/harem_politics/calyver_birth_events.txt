#Events for birth on actions

namespace = birth

birth.1180 = {
	type = character_event
	hidden = yes

	trigger = {
		any_parent = {
			is_dishu_parent_trigger = yes
			save_temporary_scope_as = dishu_parent
		}
		should_be_di_child_trigger = {
			PARENT = scope:dishu_parent
		}
	}

	immediate = {
		add_trait = di_child
		if = {
			limit = {
				mother ?= {
					is_alive = yes
					has_character_flag = should_be_di_child_flag
				}
			}
			mother = {
				remove_character_flag = should_be_di_child_flag
			}
		}
	}
}
